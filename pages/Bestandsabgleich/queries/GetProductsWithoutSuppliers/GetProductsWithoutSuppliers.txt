SELECT
    p.status,
		p.barcodes,
    p.vendor,    
    p.title,

    CASE
        WHEN p.variant = 'Default Title' THEN ''
        ELSE p.variant
    END AS variant,

    CONCAT(p.content_amount, ' ', p.content_unit) AS content,
    p.quantity,
    p.inventory_policy AS policy

FROM public.products p

WHERE
    p.barcodes IS NOT NULL
    AND cardinality(p.barcodes) > 0

    AND NOT EXISTS (
        SELECT 1
        FROM public.supplier_inventory si
        WHERE si.barcode IS NOT NULL
          AND si.barcode <> ''
          AND si.barcode = ANY(p.barcodes)
    )

    AND (
        '{{SelectStatus.selectedOptionValue}}' = ''
        OR p.status = '{{SelectStatus.selectedOptionValue}}'
    )

    AND (
        '{{SelectVendor.selectedOptionValue}}' = ''
        OR p.vendor = '{{SelectVendor.selectedOptionValue}}'
    )

    AND (
        '{{SelectCategory.selectedOptionValue}}' = ''
        OR p.category = '{{SelectCategory.selectedOptionValue}}'
    )

ORDER BY
    p.vendor,
    p.title;
