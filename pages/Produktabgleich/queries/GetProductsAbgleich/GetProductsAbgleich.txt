SELECT
		p.vendor, 
		p.barcode,
    p.title,
    p.variant,          
		CONCAT(p.content_amount, ' ', p.content_unit) AS content,
    p.quantity AS shop_quantity,
		p.inventory_tracked,
		p.inventory_policy,
    s.supplier_count,
    s.supplier_quantity,
    s.suppliers

FROM public."products" p

INNER JOIN (
    SELECT
        barcode,
        COUNT(*) AS supplier_count,
        SUM(quantity) AS supplier_quantity,

        JSON_AGG(
            JSON_BUILD_OBJECT(
                'supplier', supplier,
                'quantity', quantity
            )
            ORDER BY supplier
        ) AS suppliers

    FROM (
        SELECT
            barcode,
            supplier,
            MAX(quantity) AS quantity
        FROM public."supplier_inventory"
        GROUP BY barcode, supplier
    ) x

    GROUP BY barcode
) s
    ON p.barcode = s.barcode

WHERE p.barcode IS NOT NULL
  AND p.barcode <> ''
	
	AND
	(
  	'{{SelectVendor.selectedOptionValue}}' = ''
  	OR p.vendor = '{{SelectVendor.selectedOptionValue}}'
  )

ORDER BY p.vendor
