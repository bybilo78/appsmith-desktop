SELECT
    p.status,
    p.barcodes,
    p.vendor,
    p.title,
    p.type,
		p.supplier_sync_active,
		p.sku,

    CASE
        WHEN p.variant = 'Default Title' THEN ''
        ELSE p.variant
    END AS variant,

    CONCAT(p.content_amount, ' ', p.content_unit) AS content,
    p.quantity,
    p.inventory_policy AS policy,
    s.supplier_quantity

FROM public."products" p

LEFT JOIN LATERAL (
    SELECT
        SUM(barcode_totals.supplier_quantity) AS supplier_quantity
    FROM (
        SELECT
            si.barcode,
            SUM(si.quantity) AS supplier_quantity
        FROM public."supplier_inventory" si
        WHERE si.barcode = ANY(p.barcodes)
          AND si.barcode IS NOT NULL
          AND si.barcode <> ''
        GROUP BY si.barcode
    ) barcode_totals
) s ON TRUE

WHERE
    (
        '{{SelectStatus.selectedOptionValue}}' = ''
        OR p.status = '{{SelectStatus.selectedOptionValue}}'
    )

    AND (
        '{{SelectVendor.selectedOptionValue}}' = ''
        OR p.vendor_normalized = '{{SelectVendor.selectedOptionValue}}'
    )

    AND (
        '{{SelectCategory.selectedOptionValue}}' = ''
        OR p.category = '{{SelectCategory.selectedOptionValue}}'
    )

    AND (
        '{{SelectPolicy.selectedOptionValue}}' = ''
        OR p.inventory_policy = '{{SelectPolicy.selectedOptionValue}}'
    )

    AND (
        '{{SelectType.selectedOptionValue}}' = ''
        OR p.type = '{{SelectType.selectedOptionValue}}'
    )

ORDER BY
    p.vendor;
