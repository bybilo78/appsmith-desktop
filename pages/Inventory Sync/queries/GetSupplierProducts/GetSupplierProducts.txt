SELECT
    si.barcode,
    si.title,
    si.vendor,
    si.vendor_normalized,
    si.supplier,
    si.quantity,

    EXISTS (
        SELECT 1
        FROM public.products p
        WHERE si.barcode = ANY(p.barcodes)
    ) AS shop_product_assigned

FROM public.supplier_inventory si

WHERE
    si.barcode IS NOT NULL
    AND si.barcode <> ''

    AND (
        '{{SelectVendor.selectedOptionValue}}' = ''
        OR si.vendor_normalized = '{{SelectVendor.selectedOptionValue}}'
    )

    AND (
        '{{SelectAssigned.selectedOptionValue}}' = ''
        OR (
            EXISTS (
                SELECT 1
                FROM public.products p
                WHERE si.barcode = ANY(p.barcodes)
            )
        )::text = '{{SelectAssigned.selectedOptionValue}}'
    )

    AND (
        '{{InputSearchSupplierProducts.text}}' = ''
        OR string_to_array(
            lower(
                regexp_replace(
                    si.title,
                    '[^a-zA-Z0-9äöüÄÖÜß]+',
                    ' ',
                    'g'
                )
            ),
            ' '
        )
        @>
        string_to_array(
            lower(trim('{{InputSearchSupplierProducts.text}}')),
            ' '
        )
    )

ORDER BY
    si.vendor,
    si.title,
    si.supplier;
