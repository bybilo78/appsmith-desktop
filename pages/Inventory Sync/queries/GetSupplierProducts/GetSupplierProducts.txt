
SELECT
    si.barcode,
    si.title,
    si.vendor,
    si.vendor_normalized,
    si.supplier,
    si.quantity,
		si.price,

    EXISTS (
        SELECT 1
        FROM public.products p
        WHERE si.barcode = ANY(p.barcodes)
    ) AS shop_product_assigned

FROM public.supplier_inventory si

WHERE
    si.barcode IS NOT NULL
    AND si.barcode <> ''

    AND (
        '{{SelectVendor.selectedOptionValue}}' = ''
        OR si.vendor_normalized = '{{SelectVendor.selectedOptionValue}}'
    )

    AND (
        '{{SelectAssigned.selectedOptionValue}}' = ''
        OR (
            EXISTS (
                SELECT 1
                FROM public.products p
                WHERE si.barcode = ANY(p.barcodes)
            )
        )::text = '{{SelectAssigned.selectedOptionValue}}'
    )

    AND (
				'{{InputSearchSupplierProducts.text}}' = ''
				OR NOT EXISTS (
						SELECT 1
						FROM regexp_split_to_table(
								lower(trim('{{InputSearchSupplierProducts.text}}')),
								'\s+'
						) AS term
						WHERE term <> ''
							AND row_to_json(si)::text NOT ILIKE '%' || term || '%'
				)
		)

    AND (
        '{{RadioFilterSupplierProducts.selectedOptionValue}}' = 'all'
        OR si.barcode = ANY (
            ARRAY(
                SELECT jsonb_array_elements_text(
                    '{{ JSON.stringify(TableShopProducts.selectedRow.barcodes || []) }}'::jsonb
                )
            )
        )
    )

ORDER BY
    si.vendor,
    si.title,
    si.supplier;
