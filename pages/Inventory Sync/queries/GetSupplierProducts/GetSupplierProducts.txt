SELECT
    si.barcode,
    MAX(si.title) AS title,
    MAX(si.vendor) AS vendor,
    MAX(si.vendor_normalized) AS vendor_normalized,

    CONCAT(
        '<strong>Gesamt</strong>: ', SUM(si.quantity), ', ',
        STRING_AGG(
            CONCAT(
                '<strong>', si.supplier, '</strong>: ',
                si.quantity
            ),
            ', '
            ORDER BY si.supplier
        )
    ) AS supplier_quantity,

    EXISTS (
        SELECT 1
        FROM public.products p
        WHERE si.barcode = ANY(p.barcodes)
    ) AS shop_product_assigned

FROM (
    SELECT
        barcode,
        supplier,
        MAX(title) AS title,
        MAX(vendor) AS vendor,
        MAX(vendor_normalized) AS vendor_normalized,
        MAX(quantity) AS quantity
    FROM public.supplier_inventory
    WHERE barcode IS NOT NULL
      AND barcode <> ''
    GROUP BY
        barcode,
        supplier
) si

WHERE
    (
        '{{SelectVendor.selectedOptionValue}}' = ''
        OR si.vendor_normalized = '{{SelectVendor.selectedOptionValue}}'
    )

    AND (
        '{{SelectAssigned.selectedOptionValue}}' = ''
        OR (
            EXISTS (
                SELECT 1
                FROM public.products p
                WHERE si.barcode = ANY(p.barcodes)
            )
        )::text = '{{SelectAssigned.selectedOptionValue}}'
    )

    AND (
        '{{InputWords.text}}' = ''
        OR
        string_to_array(
            lower(
                regexp_replace(
                    si.title,
                    '[^a-zA-Z0-9äöüÄÖÜß]+',
                    ' ',
                    'g'
                )
            ),
            ' '
        )
        @>
        string_to_array(
            lower(trim('{{InputWords.text}}')),
            ' '
        )
    )

GROUP BY
    si.barcode

ORDER BY
    vendor,
    title;
