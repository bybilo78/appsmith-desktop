SELECT    
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'ACTIVE') AS active,
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'DRAFT') AS draft,
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'ARCHIVED') AS archive,    
    COUNT(DISTINCT variant_id) AS shop_variants,    
    COALESCE(SUM(quantity) FILTER (WHERE status = 'ACTIVE'), 0) AS shop_stock,		 
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'ACTIVE' AND inventory_policy = 'CONTINUE') AS policy_continue,
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'ACTIVE' AND inventory_policy = 'DENY') AS policy_deny,
		 
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
          AND NOT EXISTS (
              SELECT 1
              FROM public."supplier_inventory" si
              WHERE si.barcode IS NOT NULL
                AND si.barcode <> ''
                AND si.barcode = ANY(p.barcodes)
          )
    ) AS without_supplier_match,
		
		COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
          AND p.inventory_policy = 'DENY'
          AND p.quantity < 0
    ) AS negative_stock_policy_deny,
		
		COUNT(DISTINCT p.variant_id) FILTER (
				WHERE p.status = 'ACTIVE'
					AND (
							SELECT COUNT(DISTINCT si.supplier)
							FROM public."supplier_inventory" si
							WHERE si.barcode IS NOT NULL
								AND si.barcode <> ''
								AND si.barcode = ANY(p.barcodes)
					) = 1
		) AS active_one_supplier,
		
		COUNT(DISTINCT p.variant_id) FILTER (
				WHERE p.status = 'ACTIVE'
					AND (
							p.barcodes IS NULL
							OR NOT EXISTS (
									SELECT 1
									FROM unnest(p.barcodes) AS b(barcode)
									WHERE barcode IS NOT NULL
										AND TRIM(barcode) <> ''
							)
					)
		) AS active_without_barcode

FROM public."products" p;
