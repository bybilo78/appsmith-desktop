SELECT

    -- Alle Varianten
    COUNT(DISTINCT p.variant_id) AS variants,

    -- Status
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
    ) AS active,

    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'DRAFT'
    ) AS draft,

    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ARCHIVED'
    ) AS archive,


    -- ACTIVE: Match vorhanden
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
        AND EXISTS (
            SELECT 1
            FROM public.competitor_match_scores cms
            WHERE cms.variant_id = p.variant_id
        )
    ) AS match_count,
		
		-- ACTIVE: Matches vorhanden, aber kein einziger gültiger Match
		COUNT(DISTINCT p.variant_id) FILTER (
				WHERE p.status = 'ACTIVE'
				AND EXISTS (
						SELECT 1
						FROM public.competitor_price_analysis cpa
						WHERE cpa.variant_id = p.variant_id
							AND cpa.valid_match_count = 0
				)
		) AS invalid_matches,


    -- ACTIVE: kein Eintrag in competitor_price_analysis
		-- und Wettbewerber-Abgleich nicht deaktiviert
		COUNT(DISTINCT p.variant_id) FILTER (
				WHERE p.status = 'ACTIVE'
					AND p.competitor_sync_active IS NOT FALSE
					AND NOT EXISTS (
							SELECT 1
							FROM public.competitor_price_analysis cpa
							WHERE cpa.variant_id = p.variant_id
					)
		) AS no_competitor,
		
		-- ACTIVE: Wettbewerber-Abgleich deaktiviert
		COUNT(DISTINCT p.variant_id) FILTER (
				WHERE p.status = 'ACTIVE'
					AND p.competitor_sync_active = FALSE
		) AS competitor_sync_disabled,

    -- ACTIVE: mindestens 1 gültiger Match
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
        AND EXISTS (
            SELECT 1
            FROM public.competitor_price_analysis cpa
            WHERE cpa.variant_id = p.variant_id
              AND cpa.valid_match_count > 0
        )
    ) AS valid_matches,


    -- ACTIVE: genau 1 gültiger Competitor
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
        AND EXISTS (
            SELECT 1
            FROM public.competitor_price_analysis cpa
            WHERE cpa.variant_id = p.variant_id
              AND cpa.valid_match_count = 1
        )
    ) AS valid_match_1,


    -- ACTIVE: genau 2 gültige Competitors
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
        AND EXISTS (
            SELECT 1
            FROM public.competitor_price_analysis cpa
            WHERE cpa.variant_id = p.variant_id
              AND cpa.valid_match_count = 2
        )
    ) AS valid_match_2,


    -- ACTIVE: mehr als 2 gültige Competitors
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
        AND EXISTS (
            SELECT 1
            FROM public.competitor_price_analysis cpa
            WHERE cpa.variant_id = p.variant_id
              AND cpa.valid_match_count > 2
        )
    ) AS valid_match_more_than_2

FROM public.products p;
