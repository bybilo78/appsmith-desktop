SELECT
    COUNT(DISTINCT variant_id) AS total_variants,

    COUNT(DISTINCT variant_id) FILTER (
        WHERE final_action = 'increase'
    ) AS price_increase,

    COUNT(DISTINCT variant_id) FILTER (
        WHERE final_action = 'decrease'
    ) AS price_decrease,

    COUNT(DISTINCT variant_id) FILTER (
        WHERE final_action = 'no_change'
    ) AS price_no_change

FROM public.competitor_price_analysis;
