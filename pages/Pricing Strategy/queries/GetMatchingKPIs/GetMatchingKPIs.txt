SELECT

    -- Alle Varianten
    COUNT(DISTINCT p.variant_id) AS variants,

    -- Status
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
    ) AS active,

    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'DRAFT'
    ) AS draft,

    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ARCHIVED'
    ) AS archive,


    -- ACTIVE: Wettbewerber-Abgleich deaktiviert
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
          AND p.competitor_sync_active = FALSE
    ) AS competitor_sync_disabled,


    -- ACTIVE: Abgleich aktiv + mindestens ein valider Handle-Match
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
          AND p.competitor_sync_active IS NOT FALSE
          AND EXISTS (
              SELECT 1
              FROM unnest(p.competitor_urls) AS u(url)
              JOIN public.competitor_products cp
                ON split_part(cp.handle, CHR(63), 1) =
                   split_part(
                       split_part(u.url, '/products/', 2),
                       CHR(63),
                       1
                   )
          )
    ) AS handle_match,


    -- ACTIVE: Abgleich aktiv + kein valider Handle-Match
    COUNT(DISTINCT p.variant_id) FILTER (
        WHERE p.status = 'ACTIVE'
          AND p.competitor_sync_active IS NOT FALSE
          AND NOT EXISTS (
              SELECT 1
              FROM unnest(p.competitor_urls) AS u(url)
              JOIN public.competitor_products cp
                ON split_part(cp.handle, CHR(63), 1) =
                   split_part(
                       split_part(u.url, '/products/', 2),
                       CHR(63),
                       1
                   )
          )
    ) AS no_handle_match

FROM public.products p;
