SELECT    
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'ACTIVE') AS active,
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'DRAFT') AS draft,
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'ARCHIVED') AS archive,    
    COUNT(DISTINCT variant_id) AS shop_variants   
  
FROM public."products" p;
