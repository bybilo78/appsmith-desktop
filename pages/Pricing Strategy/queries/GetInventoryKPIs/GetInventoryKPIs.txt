SELECT    
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'ACTIVE') AS active,
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'DRAFT') AS draft,
    COUNT(DISTINCT variant_id) FILTER (WHERE status = 'ARCHIVED') AS archive,
    COUNT(DISTINCT variant_id) AS shop_variants,

    -- Aktive Varianten mit mindestens einem Competitor-Match
    COUNT(DISTINCT variant_id) FILTER (
        WHERE status = 'ACTIVE'
        AND EXISTS (
            SELECT 1
            FROM public.competitor_match_scores cms
            WHERE cms.variant_id = p.variant_id
        )
    ) AS active_with_competitor_match,

    -- Aktive Varianten ohne Competitor-Match
    COUNT(DISTINCT variant_id) FILTER (
        WHERE status = 'ACTIVE'
        AND NOT EXISTS (
            SELECT 1
            FROM public.competitor_match_scores cms
            WHERE cms.variant_id = p.variant_id
        )
    ) AS active_without_competitor_match

FROM public."products" p;
